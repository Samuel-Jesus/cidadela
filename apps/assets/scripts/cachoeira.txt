  [-.65,.65].forEach(x=>{ bx(.5,2.4,.5,m.creme,x,1.2,.1,mx); bx(.55,.1,.55,m.pedra,x,2.4,.1,mx); bx(.2,.35,.04,m.vidro,x,2.0,.37,mx); const sp=new THREE.Mesh(new THREE.CylinderGeometry(0,.38,1.1,4),new THREE.MeshStandardMaterial({map:azul,roughness:.6})); sp.rotation.y=PI/4; sp.position.set(x,3,.1); mx.add(sp); cyl(0,.03,.3,m.preto,x,3.7,.1,mx,6); });
  const fr=new THREE.Shape(); fr.moveTo(-.8,0); fr.lineTo(.8,0); fr.lineTo(0,.5); fr.closePath(); const pe=new THREE.Mesh(new THREE.ExtrudeGeometry(fr,{depth:.2,bevelEnabled:false}),m.branco); pe.position.set(0,1.7,.65); mx.add(pe);

  const arit=novo(arv,3.1,sup1(3.1)+.04,1); arit.d=.3;

  /* sobrados grandes coloridos em volta da praça, capela com cúpula */
  const grande=(x,z,w,fl,cor,rot)=>{ const gg=new THREE.Group(), h=fl*.5; bx(w,h+1,.9,M(cor),0,h/2-.5,0,gg); bx(w+.06,.05,.96,m.branco,0,h,0,gg); hip(.08,1,.45,m.telha,0,h+.22,0,(w+.2)/1.414,1.1/1.414,gg);
    for(let f=0;f<fl;f++) for(let k=0;k<Math.floor(w/.4);k++){ bx(.12,.22,.03,m.azul,-w/2+.25+k*.4,.3+f*.5,.46,gg); if(f===1) bx(.2,.03,.1,m.ferro,-w/2+.25+k*.4,.18+f*.5,.5,gg); }
    gg.rotation.y=rot; return gg; };

  [gA,gB,gC].forEach((it,i)=>it.d=.1*i);
  const cap=new THREE.Group(); bx(.9,.8,.9,m.branco,0,.4,0,cap); const dm=new THREE.Mesh(new THREE.SphereGeometry(.4,16,10,0,PI*2,0,PI/2),m.branco); dm.position.y=.8; cap.add(dm); cyl(.02,.02,.3,m.preto,0,1.35,0,cap,6); bx(.2,.4,.04,m.madeira,0,.2,.46,cap);

    for(let k=0;k<12;k++){ const x=-2.5+k*.45; bx(.04,.55,.04,m.ferro,x,.84,z,pt); const d=bx(.04,.62,.04,m.ferro,x+.22,.84,z,pt); d.rotation.z=(k%2?1:-1)*.7; } });
  [-1.3,1.3].forEach(x=>bx(.4,.6,.7,m.pedra,x,.25,0,pt));
  const pit=novo(pt,-1,0,2.2); pit.d=.4;

  /* orla, postes de luz com fios */
  const or=new THREE.Group(); bx(.5,.06,14,m.pedra,1.2,.33,0,or); for(let z=-6;z<=6.1;z+=2){ cyl(.025,.03,.7,m.ferro,1.05,.7,z,or,6); ball(.07,m.lamp,1.05,1.08,z,or); }
  const orit=novo(or,0,0,0); orit.d=.3;
  const pl=new THREE.Group(); const polos=[]; for(let z=-6;z<=6.1;z+=3){ cyl(.03,.04,1,m.madeira,4.9,sup1(4.9)+.5,z,pl,6); polos.push(z); }
  for(let k=0;k<polos.length-1;k++) [1.0,.9].forEach(h=>{ const pts=[new THREE.Vector3(4.9,sup1(4.9)+h,polos[k]),new THREE.Vector3(4.9,sup1(4.9)+h-.12,(polos[k]+polos[k+1])/2),new THREE.Vector3(4.9,sup1(4.9)+h,polos[k+1])]; pl.add(new THREE.Mesh(new THREE.TubeGeometry(new THREE.CatmullRomCurve3(pts),8,.005,3),m.preto)); });

    for(let k=1;k<n;k++){ const s=k/n, f=new THREE.Mesh(triG,cores[k%6]); f.position.set(a.x+(b.x-a.x)*s,a.y+(b.y-a.y)*s-sag*4*s*(1-s),a.z+(b.z-a.z)*s); festa.add(f); flags.push({f,ph:k}); } };
  [-3,0,3].forEach(z=>fio(new THREE.Vector3(-4.4,2.4,z),new THREE.Vector3(1.4,2.4,z),.35,22));
  [4.3,4.9].forEach(x=>fio(new THREE.Vector3(x,sup1(x)+1.3,-6.5),new THREE.Vector3(x,sup1(x)+1.3,6.5),.4,34));

  const fit=novo(fgn,3.2,sup1(3.1)+.05,2.2); fit.d=.4;

  const palit=novo(pal2,2.7,sup1(2.7)+.04,.3); palit.d=.5;

  /* gente: pontos pelas ruas e pela praça */
  const disco=tex(32,32,(c)=>{c.fillStyle='#fff';c.beginPath();c.arc(16,16,15,0,7);c.fill()});
  const N=110, hp=new Float32Array(N*3), tp=new Float32Array(N*3), hc=new Float32Array(N*3), tc=new Float32Array(N*3), gente=[], col=new THREE.Color();
  const peles=[0x6b4a36,0x8a5a3c,0xc08a60,0xf0c9a0], rou=[0xffffff,0x2fa04a,0xf2c230,0xd23a2a,0x2a62c4,0xe58aa5,0xff8a1e,0x222222];

    itens.forEach(it=>{ if(it.alvoNovo!==it.alvo){ it.atraso-=dt; if(it.atraso<=0) it.alvo=it.alvoNovo; }

    flags.forEach(f=>{ f.f.rotation.z=.3*Math.sin(t*2.4+f.ph); });
