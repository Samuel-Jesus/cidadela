// Requer Three.js. Uso: const modelo = criarFarolDaBarra(THREE); cena.add(modelo);
// No loop de animação: modelo.userData.tick(deltaSegundos);
// Botão do farol: modelo.userData.alternarLuz();

function criarFarolDaBarra(THREE){
  const g=new THREE.Group(); g.name='FarolDaBarra'; const PI=Math.PI;
  let sd=11; const rnd=()=>(sd=(sd*16807)%2147483647)/2147483647;
  const M=(c,o)=>new THREE.MeshStandardMaterial(Object.assign({color:c,roughness:.85},o||{}));
  const m={stone:M(0x756f63,{flatShading:true}),stoneL:M(0xa59f90),terr:M(0x595c5e),white:M(0xf2f2ea),roof:M(0xd2905a,{flatShading:true}),
    dark:M(0x1d1e20),win:M(0x2d4a3a),black:M(0x16181c,{roughness:.5}),glass:M(0xa8dcec,{transparent:true,opacity:.6,roughness:.15}),
    lamp:M(0xffe9a0,{emissive:0xffcf4d,emissiveIntensity:1.2}),grass:M(0x7fa33f),dirt:M(0x8c6d4c),path:M(0xcdb9ad),bush:M(0x3f7a2a),
    rock:M(0x4a3f38,{flatShading:true}),sea:M(0x1a66b0,{roughness:.3}),seaS:M(0x0f4580),shallow:M(0x35b5d0,{roughness:.3}),
    foam:M(0xffffff,{transparent:true,opacity:.6}),blue:M(0x3a6fd0),post:M(0xe8e8e8),
    beam:M(0xfff0a8,{transparent:true,opacity:.16,side:THREE.DoubleSide,depthWrite:false})};
  const bx=(w,h,d,mt,x,y,z,p)=>{const o=new THREE.Mesh(new THREE.BoxGeometry(w,h,d),mt);o.position.set(x,y,z);(p||g).add(o);return o};
  const cyl=(rt,rb,h,mt,x,y,z,seg)=>{const o=new THREE.Mesh(new THREE.CylinderGeometry(rt,rb,h,seg||16),mt);o.position.set(x,y,z);g.add(o);return o};
  const ball=(r,mt,x,y,z)=>{const o=new THREE.Mesh(new THREE.SphereGeometry(r,12,10),mt);o.position.set(x,y,z);g.add(o);return o};

  /* mar, águas rasas, espuma e terreno */
  bx(17,1,13,m.sea,-1.5,-.85,-.5).material=[m.seaS,m.seaS,m.sea,m.seaS,m.seaS,m.seaS];
  bx(12,.02,10.1,m.shallow,1,-.345,.95);
  bx(.16,.01,9.5,m.foam,-5.05,-.33,.9); bx(12,.01,.16,m.foam,1,-.33,-4.15);
  bx(11.2,1.35,9.3,m.grass,1.4,-.675,1.35).material=[m.dirt,m.dirt,m.grass,m.dirt,m.dirt,m.dirt];
  for(let i=0;i<24;i++){
    const o=new THREE.Mesh(new THREE.IcosahedronGeometry(.18+rnd()*.25,0),m.rock);
    if(i<12) o.position.set(-4.2+(rnd()-.5)*.8,-.05,-3.3+rnd()*7.5); else o.position.set(-4.2+rnd()*8.2,-.05,-3.3+(rnd()-.5)*.8);
    o.scale.y=.7; g.add(o);
  }

  /* fortaleza em polígono: muralha de pedra, terraço escuro e parapeito branco */
  const P=[[-2.8,1.5],[2.6,1.5],[2.6,-1.4],[-.2,-2.1],[-2.8,-.7]];
  const poly=k=>{const s=new THREE.Shape(); P.forEach((p,i)=>{const x=-.1+(p[0]+.1)*k, y=-(-.2+(p[1]+.2)*k); i?s.lineTo(x,y):s.moveTo(x,y)}); s.closePath(); return s};
  const pr=(shape,h,mt,y)=>{const o=new THREE.Mesh(new THREE.ExtrudeGeometry(shape,{depth:h,bevelEnabled:false}),mt); o.rotation.x=-PI/2; o.position.y=y; g.add(o)};
  const anel=(ko,ki,h,mt,y)=>{const s=poly(ko); s.holes.push(new THREE.Path(poly(ki).getPoints())); pr(s,h,mt,y)};
  anel(1,.88,1.2,m.stone,0);
  pr(poly(.9),1.15,m.terr,0);
  anel(1.01,.95,.18,m.white,1.2);

  /* portão, guaritas, quartel com telhado, anexo */
  bx(.9,.95,.08,m.stoneL,.5,.55,1.54); bx(.4,.55,.04,m.dark,.5,.4,1.585);
  const arco=new THREE.Mesh(new THREE.CircleGeometry(.2,14,0,PI),m.dark); arco.position.set(.5,.68,1.585); g.add(arco);
  [[-2.8,1.5],[2.6,1.5],[2.6,-1.4],[-2.8,-.7],[-.2,-2.1]].forEach(p=>{
    cyl(.2,.2,.32,m.stoneL,p[0],1.36,p[1],12); cyl(0,.24,.2,m.stone,p[0],1.62,p[1],12);
  });
  bx(4.3,.7,1.2,m.white,.35,1.55,.8);
  for(let i=0;i<7;i++) bx(.2,.3,.03,m.win,-1.5+i*.6,1.62,1.41);
  const tri=new THREE.Shape(); tri.moveTo(-.75,0); tri.lineTo(.75,0); tri.lineTo(0,.5);
  const tel=new THREE.Mesh(new THREE.ExtrudeGeometry(tri,{depth:4.5,bevelEnabled:false}),m.roof);
  tel.rotation.y=PI/2; tel.position.set(-1.9,1.9,.8); g.add(tel);
  bx(1.8,.55,1,m.white,1.5,1.45,-.7);

  /* torre listrada, lanterna, cúpula e facho */
  const TX=-.8,TZ=-.7,B=1.15, r=y=>.52-.12*(y/3.05);
  [[0,.8,m.white],[.8,1.75,m.black],[1.75,3.05,m.white]].forEach(b=>cyl(r(b[1]),r(b[0]),b[1]-b[0],b[2],TX,B+(b[0]+b[1])/2,TZ,24));
  cyl(.52,.44,.3,m.black,TX,B+3.2,TZ,24);
  const rail=new THREE.Mesh(new THREE.TorusGeometry(.48,.02,6,24),m.black); rail.rotation.x=PI/2; rail.position.set(TX,B+3.45,TZ); g.add(rail);
  cyl(.3,.3,.6,m.glass,TX,B+3.65,TZ,16);
  for(let i=0;i<8;i++){ const a=i*PI/4; bx(.03,.6,.03,m.white,TX+Math.cos(a)*.3,B+3.65,TZ+Math.sin(a)*.3); }
  ball(.12,m.lamp,TX,B+3.65,TZ);
  cyl(0,.36,.32,m.black,TX,B+4.11,TZ,16); ball(.05,m.black,TX,B+4.32,TZ);
  const beams=new THREE.Group(); beams.position.set(TX,B+3.65,TZ); g.add(beams);
  [0,PI].forEach(a=>{
    const piv=new THREE.Group(); piv.rotation.y=a; beams.add(piv);
    const c=new THREE.Mesh(new THREE.ConeGeometry(1,4.5,20,1,true),m.beam); c.rotation.z=PI/2; c.position.x=2.25; piv.add(c);
  });

  /* caminhos, jardim, bancos e postes */
  bx(.7,.04,3.3,m.path,.5,.02,3.15); bx(10.6,.04,.5,m.path,1.3,.02,4.8); bx(.5,.04,8.2,m.path,-3.7,.02,.9);
  [-1.9,-1,1,1.9].forEach(dx=>{ for(let i=0;i<4;i++){ const s=ball(.17,m.bush,.5+dx,.15,2.1+i*.7); s.scale.y=.9; } });
  ball(.28,m.bush,-.45,.25,1.95); ball(.28,m.bush,1.45,.25,1.95);
  bx(.7,.12,.25,m.blue,4.6,.3,4.2);
  [[-3.7,0],[-3.7,3],[2.9,4.7],[5.5,4.7]].forEach(p=>{ cyl(.025,.025,.8,m.post,p[0],.4,p[1],6); ball(.06,m.post,p[0],.82,p[1]); });

  /* barco no mar */
  const barco=new THREE.Group(); bx(.4,.07,.14,m.white,0,0,0,barco); bx(.14,.08,.1,m.white,-.05,.07,0,barco); barco.position.set(-6,-.3,-5.5); g.add(barco);

  let on=true;
  g.userData.alternarLuz=()=>{ on=!on; beams.visible=on; m.lamp.emissiveIntensity=on?1.2:0; return on; };
  let t=0;
  g.userData.tick=dt=>{
    t+=dt; if(on) beams.rotation.y+=dt*1.4;
    m.foam.opacity=.5+.3*Math.sin(t*1.6);
    barco.position.x+=dt*.5; if(barco.position.x>6) barco.position.x=-9;
  };
  return g;
}
