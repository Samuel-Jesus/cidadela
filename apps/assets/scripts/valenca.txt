// Requer Three.js. Uso: const modelo = criarValenca(THREE); cena.add(modelo);
// No loop de animação: modelo.userData.tick(deltaSegundos);
// Níveis: modelo.userData.definirNivel(n)  // 0 = só natureza (rio, mata, praia), 1 a 4 = cidade enchendo (25%, 50%, 75%, 100% das casas), 5 = melhoria máxima (orla, mirante, fonte, píer e hotel)
// Nível atual: modelo.userData.nivel()
// Pontos turísticos (para legendas): modelo.userData.pontos  // [{nome, pos:[x,y,z], nivel}]

  const g=new THREE.Group(); g.name='Valenca'; const PI=Math.PI;

  const parte=(grp,x,z,min,d,y)=>{ const it=novo(grp,x,y==null?Y0:y,z); it.d=d||0; partes.push({it,min}); return it; };
  const pontos=[]; const ponto=(nome,x,y,z,nivel)=>pontos.push({nome,pos:[x,y,z],nivel});

  /* terreno, mata ao fundo, rio Una, praia e mar */
  bx(16,.4,13.9,m.grass,0,0,-1.35);
  [[-6.5,2.2],[-3,2.8],[1,3.2],[5,2.6],[7.4,2]].forEach((h,i)=>{ const o=ball(h[1],i%2?m.hill:m.hill2,h[0],.1,-7,g); o.scale.set(1.7,.6,.5); });

  const rua=(w,d,x,z,cx)=>{ const r=new THREE.Group(); bx(w,.03,d,m.asf,0,.02,0,r); parte(r,x,z,1,0); };
  rua(12.2,.7,1.7,-1.1); rua(.7,13.2,2.1,-1.4);

  const casa=(x,z,ord)=>{

    for(let f=0;f<fl;f++){ if(f===0){ bx(.14,.3,.02,m.wood,-.2,.15,d/2+.011,gh); bx(.14,.17,.02,m.jan,.2,.3,d/2+.011,gh); } else { bx(.14,.17,.02,m.jan,-.2,.3+f*.5,d/2+.011,gh); bx(.14,.17,.02,m.jan,.2,.3+f*.5,d/2+.011,gh); } }

  const res=[[-1.3,-5.1,1.6,1.1],[4.5,2.4,1.5,1.4],[-2.2,.1,.9,.65]];
  const xs=[-2.7,-1.7,-.7,.3,1.3,2.9,3.9,4.9,5.9,6.9], zs=[-6.3,-5.3,-4.3,-3.3,-2.3,.1,1.1,2.1,3.1], cand=[];

  /* Igreja Matriz de Nossa Senhora do Amparo (nível 1) */
  const mt=new THREE.Group(); bx(2.2,1.3,1.8,m.white,0,.65,0,mt); bx(2.3,.06,1.9,m.palha,0,1.3,0,mt); hip(.1,1.55,.7,m.tile,0,1.65,0,1,.85,mt);
  [-.7,.7].forEach(x=>{ bx(.55,2.3,.55,m.white,x,1.15,.95,mt); hip(.04,.5,.55,m.blue,x,2.58,.95,1,1,mt); bx(.35,.5,.03,m.jan,x,1.9,1.23,mt); });
  bx(.4,.7,.03,m.wood,0,.35,.92,mt); bx(.1,.5,.1,m.white,0,2.1,0,mt); bx(.35,.1,.1,m.white,0,2.25,0,mt); bx(2.8,.03,1.2,m.walk,0,.02,1.9,mt);

  /* Mercado (nível 2) */
  const mc=new THREE.Group(); bx(1.6,.7,1.1,m.palha,0,.35,0,mc); bx(1.8,.08,1.3,m.red,0,.74,0,mc); [-.5,0,.5].forEach((x,i)=>bx(.3,.3,.02,M([0x2a9d8f,0xef476f,0x4cc9f0][i]),x,.4,.56,mc));
  parte(mc,-2.2,.1,2,.1); ponto('Mercado Municipal',-2.2,1.2,.1,2);

  /* barcos no rio Una: canoas (nível 0) e saveiros (nível 2) */
  const barco=(vela,cor)=>{ const b=new THREE.Group(); bx(.9,.14,.3,M(cor),0,.07,0,b); bx(.2,.12,.28,M(cor),.5,.12,0,b).rotation.z=.4;
    if(vela){ cyl(.015,.015,1,m.wood,0,.6,0,b,5); const s=new THREE.Mesh(new THREE.ShapeGeometry(new THREE.Shape().moveTo(0,0).lineTo(.6,0).lineTo(0,.9)),new THREE.MeshStandardMaterial({color:0xffffff,side:2,flatShading:true})); s.position.set(.04,.2,0); b.add(s); }
    g.add(b); return b; };
  const noRio=(o,min,v,off)=>movers.push({o,min,u:t=>{ const z=-7.5+((t*v+off)%14); o.position.set(rx(z)+Math.sin(off)*.3,.25+Math.sin(t*2+off)*.01,z); o.rotation.y=Math.atan2(-1,.2475*Math.cos(.55*z)); o.rotation.z=Math.sin(t*2+off)*.03; }});
  noRio(barco(0,0x7a4a2a),0,.5,0); noRio(barco(0,0x2f9a5a),0,.4,7); noRio(barco(1,0x5a3a20),2,.35,3.5);

  const fab=new THREE.Group(); bx(3,1,1.9,m.brick,0,.5,0,fab);
  for(let i=0;i<3;i++){ const r=bx(.95,.06,2,m.zinco,-1+i,1.1,0,fab); r.rotation.z=.4; }
  for(let k=0;k<5;k++) bx(.28,.3,.02,m.jan,-1.2+k*.6,.55,.96,fab);
  cyl(.15,.22,3,m.brick,1.1,1.5,-.55,fab,10); const S=[];
  for(let i=0;i<8;i++){ const s=new THREE.Mesh(new THREE.SphereGeometry(1,8,6),new THREE.MeshStandardMaterial({color:0xd0d0d0,transparent:true,flatShading:true})); fab.add(s); S.push(s); }
  const fit=parte(fab,-1.3,-5.1,3,.2); ponto('Companhia Valença Industrial',-1.3,3.4,-5.1,3);
  anims.push(t=>{ if(!fit.g.visible) return; S.forEach((s,i)=>{ const p=(t*.3+i/8)%1; s.position.set(1.1+p*.9,3.1+p*1.3,-.55); s.scale.setScalar(.12+p*.3); s.material.opacity=.6*(1-p); }); });
  const pt=new THREE.Group(); bx(3.2,.1,.9,m.walk,0,.08,0,pt); [-1,0,1].forEach(x=>cyl(.08,.1,.3,m.laje,x,0,0,pt,6)); [-.4,.4].forEach(z=>bx(3.2,.12,.05,m.wood,0,.2,z,pt));
  parte(pt,-5.9,-1.1,3,.1); ponto('Ponte sobre o Rio Una',-5.9,.9,-1.1,3);
  const carro=(cor)=>{ const c=new THREE.Group(); bx(.5,.12,.26,M(cor),0,.12,0,c); bx(.28,.1,.22,m.glass,-.03,.23,0,c); g.add(c); return c; };
  [[0xef476f,1,0],[0x4cc9f0,-1,3],[0xffd166,1,6]].forEach(a=>{ const o=carro(a[0]); movers.push({o,min:3,u:t=>{ const u=(t*1.4+a[2])%12; o.position.set(a[1]>0?-4.2+u:7.6-u,Y0+.02,a[1]>0?-1.25:-.95); o.rotation.y=a[1]>0?0:PI; }}); });

  movers.push({o:fe,min:3,u:t=>{ const s=Math.cos(t*.25)>0?1:-1; fe.position.set(1.8+Math.sin(t*.25)*3.6,.2+Math.sin(t*2)*.01,7.7); fe.rotation.y=s>0?0:PI; }});

  const pr=new THREE.Group(); for(let i=0;i<9;i++){ const x=-3+i*1.15, c=M([0xef476f,0xffd166,0x06d6a0,0x4cc9f0][i%4]); cyl(.015,.015,.5,m.white,x,.29,6.5,pr,5); cyl(0,.3,.12,c,x,.58,6.5,pr,8); bx(.3,.04,.14,m.white,x+.3,.06,6.5,pr); }
  parte(pr,0,0,4,.5); ponto('Praia de Guaibim',3.5,1.3,6.5,4);
  const ct=new THREE.Group(); bx(1.1,.1,.14,m.white,0,.05,-.2,ct); bx(1.1,.1,.14,m.white,0,.05,.2,ct); bx(1,.08,.5,M(0xf4a261),0,.14,0,ct); bx(.5,.2,.36,m.blue,.1,.28,0,ct); g.add(ct);
  movers.push({o:ct,min:4,u:t=>{ const p=.5+.5*Math.sin(t*.3), s=Math.cos(t*.3)>0?1:-1; ct.position.set(-2.2+p*3.6,.2+Math.sin(t*2)*.01,8.4+p*.7); ct.rotation.y=Math.atan2(-.7*s,3.6*s); }});

  parte(to,4.6,10.6,4,.3,.45); ponto('Morro de São Paulo (Ilha de Tinharé)',4.4,2.2,10.8,4);

  /* melhoria máxima (nível 5) */
  const orla=new THREE.Group(); bx(12.6,.05,.45,m.walk,1.4,.03,0,orla); for(let x=-4.4;x<=7.2;x+=1.6){ cyl(.025,.03,.7,m.dark,x,.38,-.15,orla,6); ball(.07,m.lamp,x,.76,-.15,orla); }
  parte(orla,0,5.6,5,0); ponto('Orla revitalizada',1.4,1,5.6,5);
  const mi=new THREE.Group(); cyl(.2,.32,2,m.white,0,1,0,mi,10); bx(.7,.3,.7,m.blue,0,2.15,0,mi); bx(.8,.06,.8,m.white,0,2.33,0,mi);
  const fz=new THREE.Group(); fz.position.y=2.5; mi.add(fz); const cone=new THREE.Mesh(new THREE.ConeGeometry(.25,1,10,1,true).rotateZ(-PI/2).translate(.5,0,0),new THREE.MeshBasicMaterial({color:0xffe08a,transparent:true,opacity:.35,side:2})); fz.add(cone);
  const mit=parte(mi,-3.7,4.7,5,.2); ponto('Mirante do Rio Una',-3.7,3.1,4.7,5); anims.push(t=>{ if(mit.g.visible) fz.rotation.y=t*1.5; });
  const fo=new THREE.Group(); cyl(.7,.7,.08,m.walk,0,.04,0,fo,20); cyl(.5,.5,.14,m.rio,0,.1,0,fo,16); cyl(.06,.1,.3,m.white,0,.2,0,fo,8); const J=[]; for(let i=0;i<10;i++){ const b=ball(.025,m.white,0,0,0,fo); J.push(b); }

  anims.push(t=>{ if(fot.g.visible) J.forEach((b,i)=>{ const p=(t*.8+i/10)%1, a=i*.63; b.position.set(Math.cos(a)*p*.35,.3+Math.sin(p*PI)*.4,Math.sin(a)*p*.35); }); });
  const pi=new THREE.Group(); bx(.5,.06,2.8,m.wood,0,.02,1.6,pi); for(let k=0;k<4;k++){ cyl(.03,.03,.3,m.wood,-.2,-.05,.4+k*.7,pi,5); cyl(.03,.03,.3,m.wood,.2,-.05,.4+k*.7,pi,5); }
  bx(.9,.5,.6,m.white,0,.25,-.5,pi); hip(.05,.7,.25,m.blue,0,.62,-.5,1,.8,pi); parte(pi,-3,6.2,5,.3); ponto('Porto turístico',-3,1,7.4,5);
  const ho=new THREE.Group(); bx(1.5,2,1.2,m.white,0,1,0,ho); bx(1.6,.1,1.3,m.blue,0,2.05,0,ho); for(let f=0;f<4;f++) bx(1.52,.18,1.22,m.glass,0,.35+f*.45,0,ho);
  const ht=document.createElement('canvas'); ht.width=256; ht.height=48; const hc=ht.getContext('2d'); hc.fillStyle='#1f4f9a'; hc.fillRect(0,0,256,48); hc.fillStyle='#fff'; hc.font='bold 28px Arial,sans-serif'; hc.textAlign='center'; hc.textBaseline='middle'; hc.fillText('HOTEL ORLA',128,26);
  const hs=new THREE.Mesh(new THREE.PlaneGeometry(1.1,.2),new THREE.MeshBasicMaterial({map:new THREE.CanvasTexture(ht)})); hs.position.set(0,1.8,.62); ho.add(hs);
  parte(ho,6.6,4.3,5,.2); ponto('Hotel da Orla',6.6,2.6,4.3,5);

  /* estado, níveis e animação */
  let nivel=0, t=0, vai=true;
  const frac=[0,.25,.5,.75,1.001,1.001];
  const aplicar=()=>{
    const f=frac[nivel];
    casas.forEach(it=>{ it.alvoNovo=it.ord<f?1:0; it.atraso=rnd()*.8; });
    moitas.forEach(it=>{ it.alvoNovo=nivel===5?(it.lim>.88?1:0):(it.lim>f?1:0); it.atraso=rnd()*.6; });
    partes.forEach(p=>{ p.it.alvoNovo=nivel>=p.min?1:0; p.it.atraso=p.it.d; });
    movers.forEach(mv=>mv.o.visible=nivel>=mv.min);
  };
  g.userData.definirNivel=n=>{ nivel=Math.max(0,Math.min(5,n|0)); aplicar(); return nivel; };
  g.userData.nivel=()=>nivel;
  g.userData.alternarVida=()=>(vai=!vai);
  g.userData.pontos=pontos;
  aplicar(); itens.forEach(it=>{ it.alvo=it.alvoNovo; it.s=it.alvo; it.atraso=0; it.g.visible=it.alvo>0; it.g.scale.setScalar(Math.max(.001,it.s)); it.g.position.y=it.y; });
  g.userData.tick=dt=>{
    itens.forEach(it=>{ if(it.alvoNovo!==it.alvo){ it.atraso-=dt; if(it.atraso<=0) it.alvo=it.alvoNovo; } it.v+=(it.alvo-it.s)*dt*110; it.v*=Math.exp(-dt*8); it.s+=it.v*dt; const s=Math.max(0,it.s); it.g.visible=s>.01||it.alvo>0; it.g.scale.setScalar(Math.max(.001,s)); it.g.position.y=it.y+(1-Math.min(1,s))*1.8; });
    if(!vai) return; t+=dt;
    aguas.forEach(w=>{ const p=w.p; for(let i=0;i<p.count;i++){ const x=w.b[i*3], z=w.b[i*3+2]; p.setY(i,Math.sin(x*1.2+t*1.3+z*.7)*w.a+Math.cos(z*1.5+t)*w.a*.6); } p.needsUpdate=true; });
    movers.forEach(mv=>{ if(mv.o.visible) mv.u(t); });
    palmas.forEach(p=>{ p.top.rotation.z=Math.sin(t*1.5+p.f)*.07; });
    anims.forEach(f=>f(t));
  };
  return g;
}
