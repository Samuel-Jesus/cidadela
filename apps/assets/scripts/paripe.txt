// Requer Three.js. Uso: const modelo = criarParipe(THREE); cena.add(modelo);
// No loop de animação: modelo.userData.tick(deltaSegundos);
// Níveis: modelo.userData.definirNivel(n)  // 0 = morro vazio, 1 a 4 = favela enchendo (25%, 50%, 75%, 100% das casas), 5 = resort
// Nível atual: modelo.userData.nivel()

function criarParipe(THREE){
  const g=new THREE.Group(); g.name='Paripe'; const PI=Math.PI;
  let sd=29; const rnd=()=>(sd=(sd*16807)%2147483647)/2147483647;
  const M=(c,o)=>new THREE.MeshStandardMaterial(Object.assign({color:c,roughness:.85},o||{}));
  const m={sea:M(0x2a9fb5,{roughness:.3}),seaS:M(0x0e6a80),sand:M(0xe6d3a0),dirt:M(0x8a6d52),grass:M(0x6f9a3a),asf:M(0x44464b),walk:M(0xcfcdc4),white:M(0xf4f2ea),
    bloco:M(0x9a9a96),zinco:M(0x9aa0a6,{metalness:.3,roughness:.6}),laje:M(0xb8b6ae),caixa:M(0x2a62c4),jan:M(0x23303f,{roughness:.3}),conc:M(0xa9a7a0),wood:M(0x8a6a45),
    tile:M(0xc2562f,{flatShading:true}),blue:M(0x2b66b8),pool:M(0x25a8d8,{roughness:.2}),verde:M(0x2fa04a,{flatShading:true}),trunk:M(0x7a6a52),leaf:M(0x2f8a2e),leaf2:M(0x3f9d36),
    glass:M(0x233046,{roughness:.2}),dark:M(0x23262b),pele:M(0x7a5238),pole:M(0xb9bcc2),fio:M(0x1d1d1d)};
  const bx=(w,h,d,mt,x,y,z,p)=>{const o=new THREE.Mesh(new THREE.BoxGeometry(w,h,d),mt);o.position.set(x,y,z);(p||g).add(o);return o};
  const cyl=(rt,rb,h,mt,x,y,z,p,seg)=>{const o=new THREE.Mesh(new THREE.CylinderGeometry(rt,rb,h,seg||12),mt);o.position.set(x,y,z);(p||g).add(o);return o};
  const ball=(r,mt,x,y,z,p)=>{const o=new THREE.Mesh(new THREE.SphereGeometry(r,10,8),mt);o.position.set(x,y,z);(p||g).add(o);return o};
  const hip=(r0,r1,h,mt,x,y,z,sx,sz,p)=>{const o=new THREE.Mesh(new THREE.CylinderGeometry(r0,r1,h,4),mt);o.rotation.y=PI/4;o.scale.set(sx,1,sz);o.position.set(x,y,z);(p||g).add(o);return o};
  const tubo=(pts,r)=>{const o=new THREE.Mesh(new THREE.TubeGeometry(new THREE.CatmullRomCurve3(pts.map(p=>new THREE.Vector3(p[0],p[1],p[2]))),12,r,4),m.fio);g.add(o)};
  const sup=z=>z>=.8?.3:.3+(.8-z)/7.8*5.2;

  /* morro (perfil extrudido), orla, mar */
  const sh=new THREE.Shape(); [[-9,0],[-9,5.5],[-7,5.5],[.8,.3],[5.6,.3],[5.9,.12],[7.2,.12],[7.2,0]].forEach((p,i)=>i?sh.lineTo(p[0],p[1]):sh.moveTo(p[0],p[1])); sh.closePath();
  const ter=new THREE.Mesh(new THREE.ExtrudeGeometry(sh,{depth:16,bevelEnabled:false}),[m.dirt,m.grass]); ter.rotation.y=-PI/2; ter.position.x=8; g.add(ter);
  bx(16,.03,1.3,m.sand,0,.13,6.55); bx(16,1.2,5,m.sea,0,-.6,9.7).material=[m.seaS,m.seaS,m.sea,m.seaS,m.seaS,m.seaS];
  bx(16,.04,1.3,m.asf,0,.32,5.0); bx(16,.05,.5,m.walk,0,.33,4.15); bx(16,.05,.5,m.walk,0,.33,5.75);

  /* comunidade: cada casa é um grupo com ordem de chegada; a favela começa vazia (só mata) e vai enchendo a cada nível */
  const pal=[0xf2c230,0x3a78c2,0xe58aa5,0x7bc08a,0xe8e4d8,0xe07a3a];
  const becos=new THREE.Group(), varal=new THREE.Group(); g.add(becos); g.add(varal);
  const itens=[], casas=[], moitas=[];
  const novo=(grp,x,y,z)=>{ grp.position.set(x,y,z); g.add(grp); grp.visible=false; grp.scale.setScalar(.001); const it={g:grp,y,s:0,v:0,alvo:0,alvoNovo:0,atraso:0}; itens.push(it); return it; };
  const casa=(x,z,ord)=>{
    const gh=new THREE.Group(), w=.72+rnd()*.2, d=.82, fl=1+Math.floor(rnd()*3), h=fl*.55, t=rnd();
    const mat=t<.58?M(new THREE.Color(0xc2693f).offsetHSL(0,0,(rnd()-.5)*.1)):t<.74?m.bloco:M(pal[Math.floor(rnd()*pal.length)]);
    bx(w,h+1,d,mat,0,h/2-.5,0,gh);
    if(rnd()<.55){ bx(w+.04,.05,d+.04,m.laje,0,h+.03,0,gh); if(rnd()<.55) cyl(.1,.1,.18,m.caixa,.2,h+.15,-.12,gh,10); }
    else { const r=bx(w+.08,.04,d+.08,m.zinco,0,h+.08,0,gh); r.rotation.x=.14; }
    for(let f=0;f<fl;f++){ const n=1+(rnd()<.5?1:0); for(let k=0;k<n;k++) bx(.13,.16,.02,m.jan,(n===1?0:(k?.2:-.2)),.3+f*.55,d/2+.011,gh); }
    const it=novo(gh,x,sup(z),z); it.ord=ord; casas.push(it);
  };
  for(let r=0;r<7;r++) for(let c=-7;c<=7;c++){ const x=c*.98+(rnd()-.5)*.06, z=.2-r*1+(rnd()-.5)*.05;
    if(Math.abs(x+.3)<.55||Math.abs(x-3.6)<.55||rnd()<.12) continue; casa(x,z,.55*(r/6)+.45*rnd()); }
  casas.sort((a,b)=>a.ord-b.ord); casas.forEach((it,i)=>it.ord=i/casas.length);
  [-.3,3.6].forEach(x=>{ const e=bx(.5,.05,9.4,m.laje,x,2.9,-3.1,becos); e.rotation.x=.588; for(let k=0;k<26;k++){ const z=.7-k*.3, y=sup(z)+.04; bx(.5,.03,.08,m.conc,x,y,z,becos); } });
  for(let i=0;i<60;i++){ const x=-7.3+rnd()*14.6, z=.4-rnd()*7.2, gb=new THREE.Group(), n=2+Math.floor(rnd()*2);
    for(let k=0;k<n;k++){ const rr=.22+rnd()*.2, c=new THREE.Color().setHSL(.24+rnd()*.06,.55,.22+rnd()*.1); ball(rr,M(c,{flatShading:true}),(rnd()-.5)*.3,rr*.7,(rnd()-.5)*.3,gb); }
    const it=novo(gb,x,sup(z),z); it.lim=rnd(); moitas.push(it); }
  /* poste com fios e varal */
  [[-6,4.6],[-2,4.6],[2,4.6],[6,4.6]].forEach(p=>cyl(.05,.06,2.2,m.wood,p[0],1.4,p[1],g,8));
  [[-6,-2],[-2,2],[2,6]].forEach(a=>[1.9,1.75,1.6].forEach((y,i)=>tubo([[a[0],y,4.6],[(a[0]+a[1])/2,y-.2,4.6+i*.05],[a[1],y,4.6]],.01)));
  const cores=[0xe03030,0xf2c230,0x2e6ee0,0xffffff,0x35b85a];
  for(let k=0;k<8;k++){ const z=-.4, y=sup(z)+1.4; bx(.08,.14,.01,M(cores[k%5]),1.2+k*.18,y-.04*Math.sin(k/7*PI),z+.5,varal); }

  /* muro com grafite colorido */
  const gt=document.createElement('canvas'); gt.width=512; gt.height=64; const gc=gt.getContext('2d');
  gc.fillStyle='#e8e0cc'; gc.fillRect(0,0,512,64); for(let i=0;i<22;i++){ gc.fillStyle='hsl('+Math.floor(rnd()*360)+',75%,55%)'; gc.beginPath(); gc.arc(rnd()*512,rnd()*64,8+rnd()*22,0,7); gc.fill(); }
  const mu=new THREE.Mesh(new THREE.PlaneGeometry(14,.5),new THREE.MeshStandardMaterial({map:new THREE.CanvasTexture(gt),roughness:.9})); mu.position.set(0,.6,3.88); g.add(mu); bx(14,.5,.15,m.conc,0,.55,3.8);

  /* prédios ao fundo */
  const jt=document.createElement('canvas'); jt.width=64; jt.height=64; const jc=jt.getContext('2d'); jc.fillStyle='#ddd5bd'; jc.fillRect(0,0,64,64); jc.fillStyle='#4a5560'; for(let y=6;y<64;y+=14) for(let x=6;x<64;x+=16) jc.fillRect(x,y,8,7);
  const tj=new THREE.CanvasTexture(jt); tj.wrapS=tj.wrapT=THREE.RepeatWrapping;
  [[-7,10],[-3.5,13],[0,9],[3.8,12],[7,10.5]].forEach(b=>{ const t=tj.clone(); t.needsUpdate=true; t.repeat.set(1.2,b[1]/4); bx(2,b[1],1.8,new THREE.MeshStandardMaterial({map:t,roughness:.85}),b[0],5.5+b[1]/2,-8.3); });

  /* vegetação, trânsito, barcos, gente */
  const palma=(x,z,h,y0)=>{ const p=new THREE.Group(); p.position.set(x,y0,z); g.add(p); bx(.1,h,.1,m.trunk,0,h/2,0,p);
    for(let i=0;i<8;i++){ const yw=new THREE.Group(); yw.position.y=h; yw.rotation.y=i*PI/4; p.add(yw); const tl=new THREE.Group(); tl.rotation.z=-.3-(i%2)*.25; yw.add(tl); bx(.8,.02,.14,i%2?m.leaf:m.leaf2,.4,0,0,tl); } };
  [[-7.4,2,2],[7.4,2.2,2],[-7.5,6.4,1.8],[7.4,6.3,1.8]].forEach(p=>palma(p[0],p[1],p[2],.2));
  const carro=(cor,moto)=>{ const c=new THREE.Group(); if(moto){ bx(.5,.08,.14,M(cor),0,.14,0,c); bx(.14,.2,.1,M(cor),-.05,.26,0,c); ball(.04,m.pele,-.05,.42,0,c); bx(.08,.12,.08,m.dark,-.05,.34,0,c); } else { bx(.8,.2,.4,M(cor),0,.2,0,c); bx(.45,.17,.34,M(cor),-.05,.38,0,c); bx(.4,.1,.35,m.dark,-.05,.4,0,c); } g.add(c); return c; };
  const vei=[[5.2,1,0xeeeeee,0],[4.8,-1,0xb23a3a,0],[5.2,1,0x2f6e4a,1],[4.8,-1,0x3a5fa8,1]].map((v,i)=>{ const o=carro(v[2],v[3]); o.rotation.y=v[1]>0?0:PI; return {o,z:v[0],dir:v[1],x:-8+i*4,v:1.2+i*.3}; });
  const pesca=(x,z,cor)=>{ const b=new THREE.Group(); b.position.set(x,0,z); g.add(b); bx(1.4,.25,.5,M(cor),0,.1,0,b); bx(.35,.22,.5,M(cor),.85,.18,0,b).rotation.z=.4; bx(1.2,.04,.5,m.white,0,.24,0,b); return b; };
  const barcos=[pesca(-4,9,0x2f9a5a),pesca(2,10.2,0xe8e4d8),pesca(6,8.6,0x3a78c2)];
  const gente=[[ -.3,0],[3.6,.5],[-.3,.3],[3.6,.8]].map((p,i)=>{ const f=new THREE.Group(); g.add(f); bx(.07,.14,.05,M(cores[i]),0,.1,0,f); ball(.04,m.pele,0,.21,0,f); return {f,x:p[0],u:p[1],dir:i%2?1:-1}; });

  /* resort (nível 5): hospedagem, pavilhões, piscina e bangalôs em terraços no morro */
  const rs=new THREE.Group();
  bx(5.4,2.7,1.5,m.white,-1.6,1.35,-.5,rs); for(let f=0;f<4;f++){ bx(5.4,.12,.3,m.blue,-1.6,.55+f*.62,.4,rs); bx(5.4,.18,.03,m.white,-1.6,.67+f*.62,.55,rs); for(let k=0;k<6;k++){ bx(.34,.34,.04,m.glass,-3.9+k*.9,.4+f*.62,.26,rs); } }
  hip(.1,2.7,.5,m.verde,-1.6,2.95,-.5,1.45,.9,rs);
  bx(3,1.6,1.3,m.white,4.2,.8,-.6,rs); hip(.1,1.5,.4,m.verde,4.2,1.8,-.6,1.2,.95,rs); for(let k=0;k<3;k++) bx(.4,.36,.04,m.glass,3.4+k*.8,.9,.1,rs);
  bx(3.4,.1,1.3,m.blue,-1.5,.05,1.3,rs); bx(3.2,.12,1.1,m.pool,-1.5,.06,1.3,rs);
  [[1.4,1.1],[2.2,1.4],[3,1.1]].forEach(p=>{ [[-.35,-.35],[.35,-.35],[-.35,.35],[.35,.35]].forEach(d=>bx(.05,.6,.05,m.wood,p[0]+d[0],.4,p[1]+d[1],rs)); hip(.05,.55,.35,m.verde,p[0],.88,p[1],1,1,rs); });
  [-3.6,-2.6,-.4,.6].forEach((x,i)=>{ bx(.4,.06,.2,m.white,x,.14,2.0,rs); cyl(.02,.02,.5,m.dark,x+.3,.3,2.0,rs,6); cyl(0,.32,.12,M([0xe03030,0xf2c230,0x2e86ab,0xe58aa5][i]),x+.3,.6,2.0,rs,10); });
  const rl=document.createElement('canvas'); rl.width=512; rl.height=64; const rc=rl.getContext('2d'); rc.fillStyle='#1f4f9a'; rc.fillRect(0,0,512,64); rc.fillStyle='#fff'; rc.font='bold 38px Arial,sans-serif'; rc.textAlign='center'; rc.textBaseline='middle'; rc.fillText('RESORT PARIPE',256,34);
  const rt=new THREE.Mesh(new THREE.PlaneGeometry(2.2,.28),new THREE.MeshBasicMaterial({map:new THREE.CanvasTexture(rl)})); rt.position.set(-1.6,2.3,.26); rs.add(rt);
  const rit=novo(rs,0,.3,2.4); rit.d=0;
  const partes=[rit];
  const fileira=(z,d)=>{ const gp=new THREE.Group();
    for(let x=-5.5;x<=5.6;x+=2.2){ bx(.95,.05,.95,m.wood,x,.03,0,gp); bx(.7,.45,.7,m.white,x,.3,0,gp); hip(.05,.55,.3,m.verde,x,.7,0,1,1,gp); bx(.2,.2,.02,m.glass,x,.32,.36,gp);
      if(Math.abs(x)<4){ const p2=new THREE.Group(); p2.position.set(x+.8,0,.2); gp.add(p2); bx(.06,.7,.06,m.trunk,0,.35,0,p2); for(let k=0;k<5;k++){ const yw=new THREE.Group(); yw.position.y=.7; yw.rotation.y=k*PI*2/5; p2.add(yw); bx(.45,.02,.1,m.leaf,.22,0,0,yw); } } }
    const it=novo(gp,0,sup(z),z); it.d=d; partes.push(it); };
  fileira(-1.3,.35); fileira(-2.9,.7); fileira(-4.5,1.05);
  const pl2=new THREE.Group(); bx(4,.1,1.6,m.blue,0,.05,0,pl2); bx(3.8,.12,1.4,m.pool,0,.06,0,pl2);
  [-1.4,-.5,.5,1.4].forEach((x,i)=>{ bx(.4,.06,.2,m.white,x,.14,1.1,pl2); cyl(.02,.02,.5,m.dark,x+.3,.3,1.1,pl2,6); cyl(0,.3,.12,M([0xe03030,0xf2c230,0x2e86ab,0xe58aa5][i]),x+.3,.6,1.1,pl2,10); });
  const pit=novo(pl2,0,sup(-6),-6); pit.d=1.4; partes.push(pit);

  let nivel=0, t=0, vai=true;
  const frac=[0,.25,.5,.75,1.001,0];
  const aplicar=()=>{
    const f=frac[nivel];
    casas.forEach(it=>{ it.alvoNovo=(nivel<=4&&it.ord<f)?1:0; it.atraso=nivel===5?rnd()*.5:rnd()*.8; });
    moitas.forEach(it=>{ it.alvoNovo=nivel===5?(it.lim>.6?1:0):(it.lim>f?1:0); it.atraso=rnd()*.6; });
    partes.forEach(it=>{ it.alvoNovo=nivel===5?1:0; it.atraso=it.d; });
    const vivo=nivel>=1&&nivel<=4; becos.visible=vivo; varal.visible=nivel>=2&&nivel<=4; gente.forEach(q=>q.f.visible=vivo);
  };
  g.userData.definirNivel=n=>{ nivel=Math.max(0,Math.min(5,n|0)); aplicar(); return nivel; };
  g.userData.nivel=()=>nivel;
  g.userData.alternarVida=()=>(vai=!vai);
  aplicar(); itens.forEach(it=>{ it.alvo=it.alvoNovo; it.s=it.alvo; it.atraso=0; it.g.visible=it.alvo>0; it.g.scale.setScalar(Math.max(.001,it.s)); it.g.position.y=it.y; });
  g.userData.tick=dt=>{
    itens.forEach(it=>{ if(it.alvoNovo!==it.alvo){ it.atraso-=dt; if(it.atraso<=0) it.alvo=it.alvoNovo; } it.v+=(it.alvo-it.s)*dt*110; it.v*=Math.exp(-dt*8); it.s+=it.v*dt; const s=Math.max(0,it.s); it.g.visible=s>.01||it.alvo>0; it.g.scale.setScalar(Math.max(.001,s)); it.g.position.y=it.y+(1-Math.min(1,s))*1.8; });
    if(!vai) return; t+=dt;
    vei.forEach(v=>{ v.x+=v.dir*v.v*dt; if(v.x>9) v.x=-9; if(v.x<-9) v.x=9; v.o.position.set(v.x,.35,v.z); });
    barcos.forEach((b,i)=>{ b.position.y=.02*Math.sin(t*1.6+i*2); b.rotation.z=.03*Math.sin(t*1.3+i); });
    gente.forEach(p=>{ p.u+=p.dir*dt*.06; if(p.u>1){p.u=1;p.dir=-1} if(p.u<0){p.u=0;p.dir=1} const z=.7-p.u*7.4; p.f.position.set(p.x,sup(z)+.06+.015*Math.abs(Math.sin(t*9+p.x)),z); });
  };
  return g;
}
