// Requer Three.js. Uso: const modelo = criarPedagioLinhaVerde(THREE); cena.add(modelo);
// No loop de animação: modelo.userData.tick(deltaSegundos);
// Pausar/retomar o tráfego: modelo.userData.alternarTrafego();
// Abrir/fechar a faixa k (0 a 9): modelo.userData.alternarFaixa(k);

function criarPedagioLinhaVerde(THREE){
  const g=new THREE.Group(); g.name='PedagioLinhaVerde'; const PI=Math.PI;
  let sd=3; const rnd=()=>(sd=(sd*16807)%2147483647)/2147483647;
  const M=(c,o)=>new THREE.MeshStandardMaterial(Object.assign({color:c,roughness:.85},o||{}));
  const m={grass:M(0x7fa83c),dirt:M(0x8a6d4c),asf:M(0x4b4d52,{roughness:.95}),conc:M(0xbdbbb2),conc2:M(0xb4b2a9),roofW:M(0xf1f2f0),roofB:M(0x2b5db8),yellow:M(0xf2c230),
    cab:M(0xd8dadd),win:M(0x1f2c40,{roughness:.2}),dark:M(0x23262b),rail:M(0xc4c8cc,{metalness:.4,roughness:.5}),orange:M(0xf07b2a),white:M(0xf4f4f2),
    verde:M(0x35e06a,{emissive:0x22cc55,emissiveIntensity:1.1}),verm:M(0xff3b30,{emissive:0xdd1a10,emissiveIntensity:1.1}),hill:M(0x6f9a3a),trunk:M(0x7a6a52),leaf:M(0x2f8a2e),leaf2:M(0x3f9d36),red:M(0xd13030)};
  const bx=(w,h,d,mt,x,y,z,p)=>{const o=new THREE.Mesh(new THREE.BoxGeometry(w,h,d),mt);o.position.set(x,y,z);(p||g).add(o);return o};
  const cyl=(rt,rb,h,mt,x,y,z,seg)=>{const o=new THREE.Mesh(new THREE.CylinderGeometry(rt,rb,h,seg||10),mt);o.position.set(x,y,z);g.add(o);return o};
  const tex=(w,h,fn)=>{const c=document.createElement('canvas');c.width=w;c.height=h;fn(c.getContext('2d'),w,h);return new THREE.CanvasTexture(c)};
  const plano=(w,h,t,x,y,z,ry)=>{const p=new THREE.Mesh(new THREE.PlaneGeometry(w,h),new THREE.MeshStandardMaterial({map:t,roughness:.9}));p.position.set(x,y,z);if(ry)p.rotation.y=ry;g.add(p);return p};

  /* terreno, pista e marcações */
  bx(22,.6,20,m.grass,0,-.3,0).material=[m.dirt,m.dirt,m.grass,m.dirt,m.dirt,m.dirt];
  bx(14,.04,18,m.asf,0,.02,0);
  const linhas=tex(560,260,(c,W,H)=>{
    c.fillStyle='#4b4d52'; c.fillRect(0,0,W,H); c.fillStyle='#f2f2ee';
    for(let i=0;i<11;i++){ const x=(-6+i*1.2+7)*40; for(let y=0;y<H;y+=24) c.fillRect(x-2,y,4,12); }
    c.fillRect(6,0,3,H); c.fillRect(W-9,0,3,H);
  });
  [5.75,-5.75].forEach(z=>{ const p=plano(14,6.5,linhas,0,.045,z); p.rotation.x=-PI/2; });

  /* cobertura: fascias, laje, painéis azul e branco, logo */
  bx(12.8,.5,.4,m.conc2,0,2.35,1.5); bx(12.8,.5,.4,m.conc2,0,2.35,-1.5); bx(12.8,.12,3.4,m.conc,0,2.66,0);
  bx(7.6,.03,3.4,m.roofW,0,2.73,0); [-5.1,5.1].forEach(x=>bx(2.6,.03,3.4,m.roofB,x,2.73,0));
  const logo=tex(256,56,(c,W,H)=>{c.fillStyle='#f4f4f0';c.fillRect(0,0,W,H);c.fillStyle='#0f8a5a';c.font='bold 40px Arial,sans-serif';c.textAlign='left';c.textBaseline='middle';c.fillText('CLN',16,30);
    c.fillStyle='#2b5db8';c.font='11px Arial,sans-serif';c.fillText('Concessionária',100,22);c.fillText('Litoral Norte',100,36)});
  plano(1.8,.4,logo,-4.9,2.35,1.71);

  /* ilhas amarelas com cabines, pilares, semáforos e placas de faixa */
  const chev=tex(64,32,(c)=>{c.fillStyle='#f2c230';c.fillRect(0,0,64,32);c.strokeStyle='#111';c.lineWidth=8;for(let y=0;y<32;y+=16){c.beginPath();c.moveTo(0,y);c.lineTo(32,y+12);c.lineTo(64,y);c.stroke()}});
  for(let i=0;i<11;i++){
    const x=-6+i*1.2;
    bx(.55,.3,2.6,m.yellow,x,.19,0); plano(.5,.28,chev,x,.22,1.31);
    bx(.38,.85,.9,m.cab,x,.77,-.45); bx(.3,.3,.02,m.win,x,.9,.01);
    if(i%2===0) bx(.28,1.85,.28,m.conc,x,1.22,.65);
  }
  const lampadas=[];
  for(let k=0;k<10;k++){
    const x=-5.4+k*1.2;
    const num=tex(64,64,(c,W,H)=>{c.fillStyle='#1d4ed8';c.fillRect(0,0,W,H);c.strokeStyle='#fff';c.lineWidth=3;c.strokeRect(3,3,W-6,H-6);c.fillStyle='#fff';c.font='bold 40px Arial,sans-serif';c.textAlign='center';c.textBaseline='middle';c.fillText(String(k+1),W/2,H/2+2)});
    plano(.34,.34,num,x,2.5,1.71);
    bx(.3,.3,.05,m.dark,x,2.17,1.72);
    const l=bx(.2,.2,.06,m.verde,x,2.17,1.74); lampadas.push(l);
  }

  /* trânsito: carros e caminhão nas 10 faixas */
  const cores=[0xeeeeee,0xc6c9cd,0x23262b,0xb23a3a,0x3a5fa8,0xd7d9dc,0x2f6e4a,0xe6e1d0];
  const carro=(cor,tipo)=>{
    const c=new THREE.Group();
    if(tipo==='caminhao'){
      bx(.8,.9,2.2,M(0xe3e3de),0,.65,.35,c); bx(.8,.75,.7,M(0x3a6fb0),0,.58,-1.1,c); bx(.6,.2,.02,m.win,0,.85,-1.46,c);
    } else {
      bx(.62,.2,1.5,M(cor),0,.2,0,c); bx(.54,.2,.8,M(cor),0,.4,.1,c); bx(.5,.12,.82,m.win,0,.42,.1,c);
      [-.2,.2].forEach(x=>bx(.12,.06,.02,m.red,x,.24,.76,c));
    }
    [-.32,.32].forEach(x=>[-.5,.5].forEach(z=>bx(.08,.2,.22,m.dark,x,.1,z,c)));
    c.position.y=.05; g.add(c); return c;
  };
  const faixas=[], frota=[];
  for(let k=0;k<10;k++){
    faixas.push({x:-5.4+k*1.2,aberta:true,t:3+rnd()*10});
    frota.push({o:carro(cores[k%8],k===2?'caminhao':'carro'),k,z:3+k*1.1+rnd()*2,servido:false,espera:0});
    frota[k].o.position.x=faixas[k].x;
  }
  const lig=(k)=>{ lampadas[k].material=faixas[k].aberta?m.verde:m.verm; };

  /* guard-rails, cones, placa de cobrança automática, postes, palmeiras, morros */
  [-7.3,7.3].forEach(x=>[[-9,-3.5],[3.5,9]].forEach(r=>{
    const z=(r[0]+r[1])/2, len=r[1]-r[0];
    bx(.06,.18,len,m.rail,x,.35,z); bx(.06,.18,len,m.rail,x*.999,.18,z);
    for(let zz=r[0]+.3;zz<=r[1];zz+=.8) bx(.05,.4,.05,m.dark,x,.22,zz);
  }));
  const cone=(x,z)=>{cyl(.02,.1,.38,m.orange,x,.24,z,12);cyl(.06,.075,.06,m.white,x,.24,z,12)};
  [[6.5,3.2],[6.8,4.3],[7,5.4],[-6.5,3.2],[-6.8,4.3],[-7,5.4]].forEach(p=>cone(p[0],p[1]));
  const placa=tex(256,160,(c,W,H)=>{c.fillStyle='#1d4ed8';c.fillRect(0,0,W,H);c.strokeStyle='#fff';c.lineWidth=4;c.strokeRect(5,5,W-10,H-10);c.fillStyle='#fff';c.font='bold 28px Arial,sans-serif';c.textAlign='center';c.fillText('COBRANÇA',W/2,48);c.fillText('AUTOMÁTICA',W/2,82);c.fillRect(100,100,56,32);c.fillStyle='#1d4ed8';c.fillRect(108,108,40,10)});
  cyl(.05,.05,2.8,m.dark,8,1.4,5.5,8); bx(.06,.06,1.2,m.dark,7.5,2.7,5.5); plano(1.2,.75,placa,7.5,2.35,5.52);
  [[-8.3,-4],[8.3,-4],[-8.3,4],[8.3,4]].forEach(p=>{ cyl(.04,.05,4,m.rail,p[0],2,p[1],8); bx(.5,.05,.05,m.rail,p[0]+(p[0]<0?.25:-.25),4,p[1]); bx(.2,.06,.12,m.white,p[0]+(p[0]<0?.5:-.5),3.95,p[1]); });
  const palma=(x,z,h)=>{
    const p=new THREE.Group(); p.position.set(x,0,z); g.add(p); bx(.1,h,.1,m.trunk,0,h/2,0,p);
    for(let i=0;i<8;i++){ const yw=new THREE.Group(); yw.position.y=h; yw.rotation.y=i*PI/4; p.add(yw); const tl=new THREE.Group(); tl.rotation.z=-.3-(i%2)*.25; yw.add(tl); bx(.8,.02,.15,i%2?m.leaf:m.leaf2,.4,0,0,tl); }
  };
  [[-9.2,2.5,1.4],[9.3,-1.5,1.6],[-9.4,-2,1.2],[9.4,7,1.3]].forEach(p=>palma(p[0],p[1],p[2]));
  [[-8.5,-7],[8.7,-6.5]].forEach(p=>{ const h=new THREE.Mesh(new THREE.SphereGeometry(2.3,20,12),m.hill); h.scale.set(1.5,.6,1.2); h.position.set(p[0],0,p[1]); g.add(h); });

  /* animação: carros param na cabine, faixas alternam entre aberta e fechada */
  let vai=true;
  g.userData.alternarTrafego=()=>(vai=!vai);
  g.userData.alternarFaixa=k=>{ faixas[k].aberta=!faixas[k].aberta; lig(k); return faixas[k].aberta; };
  g.userData.tick=dt=>{
    if(!vai) return;
    faixas.forEach((f,k)=>{ f.t-=dt; if(f.t<=0){ f.t=6+rnd()*10; const abertas=faixas.filter(q=>q.aberta).length; if(f.aberta&&abertas>7) f.aberta=false; else f.aberta=true; lig(k); } });
    frota.forEach(c=>{
      const f=faixas[c.k];
      const alvo=c.servido?-99:((f.aberta||c.z<3.6)?1.9:3.7);
      if(c.z>alvo) c.z=Math.max(alvo,c.z-(c.servido?2.6:2)*dt);
      if(!c.servido&&c.z<=1.91&&f.aberta){ c.espera+=dt; if(c.espera>1.5) c.servido=true; }
      if(c.servido&&c.z<-9){ c.z=9+rnd()*4; c.servido=false; c.espera=0; }
      c.o.position.z=c.z;
    });
  };
  return g;
}
